<script>
    let counter = 0;

    function showProduct(title, price, img, desc) {
        document.getElementById('mTitle').innerText = title;
        document.getElementById('mPrice').innerText = '₪' + price;
        document.getElementById('mImg').src = img;
        document.getElementById('mDesc').innerText = desc;
        document.getElementById('pModal').classList.add('active');
    }

    function closeP() {
        document.getElementById('pModal').classList.remove('active');
    }

    function buy() {
        counter++;
        document.getElementById('count').innerText = counter;
        alert("המוצר נוסף בהצלחה!");
        closeP();
    }

    // סגירת מודאל בלחיצה מחוץ לקופסה
    window.onclick = function(e) {
        if(e.target.id === 'pModal') closeP();
    }
</script>
